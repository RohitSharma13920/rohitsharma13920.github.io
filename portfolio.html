01

Sparse Vector Spaces

Rather than deploying compute-heavy, memory-leaking transformer models for lightweight classification, we utilize custom TF-IDF feature extractions with unigram and bigram n-grams spanning a 50,000 dimension vector space.

02

Sub-15ms Asynchrony

Synchronous Flask or Django gateways choke under concurrent request bursts. We engineer on FastAPI with Uvicorn worker pools, handling model serialization and Pydantic validation via asynchronous coroutines.

■ The Engineering Challenge

Processing noisy, sarcastic, and colloquial long-form movie reviews (50,000 IMDB dataset) requires separating contextual vocabulary from narrative punctuation while keeping inference latency fast enough for real-time speech dictation.

Deployed Scikit-Learn TF-IDF vectorizers (50,000 max features, sublinear term-frequency scaling, n-gram ranges (1,2)) coupled with an L2-regularized Logistic Regression classifier. Serialized via Joblib and served via FastAPI async endpoints.

# pipeline.py — High-Throughput Scikit-Learn NLP Architecture
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
import re, joblib

class SentimentPipeline:
    def __init__(self, model_path="artifacts/model.joblib", vec_path="artifacts/vec.joblib"):
        self.vectorizer = joblib.load(vec_path)
        self.classifier = joblib.load(model_path)

    def clean_text(self, raw_text: str) -> str:
        # Strip HTML tags, non-alphabetics, lowercase normalization
        text = re.sub(r'<[^>]+>', ' ', raw_text)
        text = re.sub(r'[^a-zA-Z\s]', '', text)
        return text.lower().strip()

    def predict(self, text: str):
        cleaned = self.clean_text(text)
        tfidf_tensor = self.vectorizer.transform([cleaned])  # 50,000 N-Gram dimensions
        prob = self.classifier.predict_proba(tfidf_tensor)[0]
        prediction = "Positive" if prob[1] >= 0.5 else "Negative"
        return {"sentiment": prediction, "confidence": round(float(prob[1]), 4)}
                        
# main.py — Asynchronous Low-Latency FastAPI Microservice
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pipeline import SentimentPipeline
from schemas import ReviewRequest, PredictionResponse
import time

app = FastAPI(title="CineScope AI Telemetry Gateway", version="2.1.0")
pipeline = SentimentPipeline()

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["POST", "GET"]
)

@app.post("/api/predict", response_model=PredictionResponse)
async def infer_sentiment(payload: ReviewRequest):
    start_time = time.perf_counter()
    result = pipeline.predict(payload.review_text)
    latency_ms = (time.perf_counter() - start_time) * 1000
    
    return PredictionResponse(
        sentiment=result["sentiment"],
        confidence=result["confidence"],
        latency_ms=round(latency_ms, 2)
    )
                        
# schemas.py — Pydantic Data Contracts
from pydantic import BaseModel, Field

class ReviewRequest(BaseModel):
    review_text: str = Field(..., min_length=3, max_length=10000, description="Raw input review string")

class PredictionResponse(BaseModel):
    sentiment: str
    confidence: float
    latency_ms: float
